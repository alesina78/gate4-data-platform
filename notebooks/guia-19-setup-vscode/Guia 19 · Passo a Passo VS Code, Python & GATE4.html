3. Implementação dos Módulos Python & Pytest

Crie o pacote da aplicação e sua respectiva suíte de testes unitários automatizados.

3.1 Módulo Principal — src/config.py 
"""
Módulo de configurações centrais da plataforma GATE4.
"""
import os

class GATE4Config:
    PROJECT_NAME: str = "GATE4 Data Platform"
    ENV: str = os.getenv("GATE4_ENV", "development")
    DATA_PATH: str = os.getenv("GATE4_DATA_PATH", "./data")

    @classmethod
    def get_status(cls) -> dict:
        return {
            "project": cls.PROJECT_NAME,
            "environment": cls.ENV,
            "status": "ready"
        }
3.2 Suíte de Testes — tests/test_config.py 
from src.config import GATE4Config

def test_gate4_config_default_values():
    """Valida se as configurações padrão do GATE4 estão corretas."""
    assert GATE4Config.PROJECT_NAME == "GATE4 Data Platform"
    assert GATE4Config.ENV == "development"

def test_gate4_status_structure():
    """Valida o retorno do método get_status."""
    status = GATE4Config.get_status()
    assert isinstance(status, dict)
    assert status["status"] == "ready"
